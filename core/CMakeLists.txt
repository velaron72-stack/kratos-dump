cmake_minimum_required(VERSION 3.18)
project(kratos_core LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)

# Pinned third-party sources live in ../third_party (see DEPENDENCIES.md).
set(KRATOS_THIRD_PARTY "${CMAKE_CURRENT_LIST_DIR}/../third_party" CACHE PATH "Third-party source root")

if(NOT TARGET zlibstatic)
  set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${KRATOS_THIRD_PARTY}/zlib" "${CMAKE_BINARY_DIR}/tp_zlib" EXCLUDE_FROM_ALL)
endif()

if(NOT TARGET libzstd_static)
  set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_CONTRIB OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
  set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
  add_subdirectory("${KRATOS_THIRD_PARTY}/zstd/build/cmake" "${CMAKE_BINARY_DIR}/tp_zstd" EXCLUDE_FROM_ALL)
endif()

if(NOT TARGET capstone_static)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_CSTOOL OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_BUILD_CSTEST OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_INSTALL OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
  set(CAPSTONE_ARM_SUPPORT ON CACHE BOOL "" FORCE)
  set(CAPSTONE_ARM64_SUPPORT ON CACHE BOOL "" FORCE)
  set(CAPSTONE_X86_SUPPORT ON CACHE BOOL "" FORCE)
  add_subdirectory("${KRATOS_THIRD_PARTY}/capstone" "${CMAKE_BINARY_DIR}/tp_capstone" EXCLUDE_FROM_ALL)
endif()

file(GLOB KRATOS_CORE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")

add_library(kratos_core STATIC ${KRATOS_CORE_SOURCES})
target_include_directories(kratos_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
# ELFIO is header-only and used only as an independent cross-check parser.
target_include_directories(kratos_core PRIVATE "${KRATOS_THIRD_PARTY}/ELFIO")
target_compile_definitions(kratos_core PUBLIC _FILE_OFFSET_BITS=64)
target_compile_options(kratos_core PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
target_link_libraries(kratos_core PUBLIC zlibstatic libzstd_static capstone_static)

if(NOT ANDROID)
  find_package(Threads REQUIRED)
  target_link_libraries(kratos_core PUBLIC Threads::Threads)
endif()

# JNI bridge for the Android app (and for host tests that load it into a JVM).
option(KRATOS_BUILD_JNI "Build the JNI bridge" OFF)
if(ANDROID OR KRATOS_BUILD_JNI)
  if(NOT ANDROID)
    find_package(JNI REQUIRED)
  endif()
  add_library(kratos_jni SHARED jni/kratos_jni.cpp)
  target_link_libraries(kratos_jni PRIVATE kratos_core)
  if(ANDROID)
    target_link_libraries(kratos_jni PRIVATE log)
    # Android 15+ devices may use 16 KiB pages: align the load segments accordingly.
    target_link_options(kratos_jni PRIVATE -Wl,-z,max-page-size=16384)
  else()
    target_include_directories(kratos_jni PRIVATE ${JNI_INCLUDE_DIRS})
  endif()
  target_compile_options(kratos_jni PRIVATE -Wall -Wextra)
endif()

# Host tools and tests (not built for Android).
if(NOT ANDROID)
  option(KRATOS_BUILD_TESTS "Build host tests and CLI" ON)
  if(KRATOS_BUILD_TESTS)
    add_executable(kratos_cli tools/kratos_cli.cpp)
    target_link_libraries(kratos_cli PRIVATE kratos_core)
    file(GLOB KRATOS_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
    add_executable(kratos_tests ${KRATOS_TEST_SOURCES})
    target_link_libraries(kratos_tests PRIVATE kratos_core)
    target_compile_definitions(kratos_tests PRIVATE KRATOS_FIXTURE_DIR="${CMAKE_CURRENT_LIST_DIR}/tests/fixtures" KRATOS_SOURCE_DIR="${CMAKE_CURRENT_LIST_DIR}")
    enable_testing()
    add_test(NAME kratos_tests COMMAND kratos_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
      add_test(NAME jni_signature_consistency COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../scripts/check_jni_names.py)
    endif()
  endif()
endif()
